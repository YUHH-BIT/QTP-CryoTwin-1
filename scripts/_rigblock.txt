/* -------------------- 灾害跟拍机位（camera rig） --------------------
   【为什么重写】
   原来两套跟拍都是「相机固定在水头后方 1650 m、侧移 780 m、抬高 620 m」。实测：
     t=120 → 俯角 27.3°、相机到水头 2281 m、水体只占画面高 18.6%
     t=260 → 俯角 24.2°、相机到水头 2179 m、水体只占画面高 27.8%
     t=360 → 俯角 26.4°、相机到水头 2249 m、水体只占画面高 21.5%
   而且水面在画面里的横向范围可以到 -2.73 NDC（视口是 -1~1），
   大半条水带在框外，中间是一整面对岸山坡 —— 这就是「看不到灾害」。

   【按什么改的】
   1. 取景距离由主体包围球反解，不写死 offset：
        d ≈ R / sin(θ_fov · 占高 / 2)
      XNA 官方 FitCameraToScene、Unity/UE 的 auto-framing、three.js 社区的
      adaptCameraDistance 都是这条式子。洪水主体从 100 m 量级变到 800 m 量级，
      没有这个反解就不可能全程保持同一构图。
   2. 俯角取 40°：沟两侧 1 km 内山体比水面高 600–1000 m（实测），
      25° 的平视只会拍到对岸山坡。
   3. 阻尼只作用在 rig 参数（锚点 / 航向 / 距离）上，相机 = f(rig)。
      不做「相机世界坐标 lerp」：那样前锋一加速相机就被甩在后面，
      水直接从画面下方跑出去。
   4. 航向用 ±700 m 基线求，再与 22 km 主轴混合 30%，并做时间低通。
      chanAt 是折线（顶点间距中位数 63 m），±45 m 差分在每个顶点跳一次 ——
      这是「镜头晃动」的物理来源。
   5. 相机自身穿地只做「抬高度」兜底，不按整条视线抬：沟侧山脊会把机位
      推到山顶，谷底洪水反而看不见（上一轮已踩过这个坑）。
   6. 进入跟拍用摇臂过渡（follow 1.8 s / 航拍 1.2 s），不再瞬间瞬移。
   ------------------------------------------------------------------ */
const RIG = {
  // 播放跟拍：把整条可见水带装进画面，留出下游方向
  follow: { elev: 40, az: 20, occ: 0.62, lead: 320, body: 1400,
            minD: 640, maxD: 3800, anchorRate: 2.4, yawRate: 1.6, distRate: 2.2, intro: 1.8 },
  // 洪水航拍：更贴、更斜，看泡沫与碎冰
  cine:   { elev: 28, az: 32, occ: 0.86, lead: 220, body: 900,
            minD: 360, maxD: 2400, anchorRate: 2.8, yawRate: 2.0, distRate: 3.0, intro: 1.2 },
};
const rig = {
  mode: null,
  anchor: new THREE.Vector3(),
  yaw: 0, dist: 1500, introT: 1,
  cam0: new THREE.Vector3(), tgt0: new THREE.Vector3(),
};
const rigA = new THREE.Vector3(), rigPos = new THREE.Vector3();
const rigAim = new THREE.Vector3(), rigMount = new THREE.Vector3();
const rigTmp = new THREE.Vector3();
const rigPool = [], rigPts = [];
// 最短角差：把角度插值做成不跨 ±π 的连续量
const angDiff = (from, to) => {
  let d = (to - from) % (Math.PI * 2);
  if (d > Math.PI) d -= Math.PI * 2;
  if (d < -Math.PI) d += Math.PI * 2;
  return d;
};
/* 采样「可见灾害体」：水头往上游 body 米内的湿区（横向取 ±110 m 三条线）。
   失稳前没有水面时退化为沟道线，保证一开演就有稳定构图，不会因为
   「没有湿点」而让机位掉回上一个位置。 */
function rigBody(cfg) {
  rigPts.length = 0;
  if (!flood) return rigPts;
  const L = channel.lengthM;
  const s0 = clamp(flood.frontS || 0, 0, L);
  const sA = clamp(s0 + cfg.lead, 0, L);
  const sB = clamp(s0 - cfg.body, 0, L);
  const N = 12;
  const put = (x, y, z) => {
    let v = rigPool[rigPts.length];
    if (!v) v = rigPool[rigPts.length] = new THREE.Vector3();
    rigPts.push(v.set(x, y, z));
  };
  for (let i = 0; i <= N; i++) {
    const s = lerp(sB, sA, i / N);
    const c = flood.chanAt(s), t = flood.tangentAt(s);
    for (let k = -1; k <= 1; k++) {
      const off = k * 110;
      const x = c.x - t.z * off, z = c.z + t.x * off;
      const w = flood.waterY(x, z, flood.tField);
      if (w.h > 1.5) put(x, w.y, z);
    }
  }
  if (rigPts.length < 4) {
    rigPts.length = 0;
    for (let i = 0; i <= N; i++) {
      const s = lerp(sB, sA, i / N);
      const c = flood.chanAt(s);
      put(c.x, groundAt(c.x, c.z) + 8, c.z);
    }
  }
  return rigPts;
}
/* 每帧解一次机位：锚点/航向/距离各自阻尼，机位由它们算出来。 */
function updateFollowRig(dt, mode) {
  const cfg = RIG[mode];
  if (!flood) return;
  const h = flood.frontHint();
  if (!h || !Number.isFinite(h.x) || !Number.isFinite(h.z)) return;
  const pts = rigBody(cfg);
  if (pts.length < 2) return;
  const L = channel.lengthM;
  const s0 = clamp(flood.frontS || 0, 0, L);

  // 主体质心与包围半径
  rigA.set(0, 0, 0);
  for (const p of pts) rigA.add(p);
  rigA.multiplyScalar(1 / pts.length);
  let R = 0;
  for (const p of pts) R = Math.max(R, p.distanceTo(rigA));
  R = Math.max(R, 60);

  // 航向：局部基线（±700 m）与全链条主轴按角度混合，避免急弯处甩镜头
  const cA = flood.chanAt(clamp(s0 - 700, 0, L));
  const cB = flood.chanAt(clamp(s0 + 700, 0, L));
  const cSrc = flood.chanAt(0), cDst = flood.chanAt(L);
  const yawLocal = Math.atan2(cB.x - cA.x, cB.z - cA.z);
  const yawAxis = Math.atan2(cDst.x - cSrc.x, cDst.z - cSrc.z);
  const yawT = yawLocal + angDiff(yawLocal, yawAxis) * 0.30;

  const entering = rig.mode !== mode;
  const snap = entering || (mode === 'cine' ? cineSnap : followSnap);
  if (mode === 'cine') cineSnap = false; else followSnap = false;
  if (entering) {
    rig.mode = mode;
    rig.cam0.copy(camera.position);
    rig.tgt0.copy(controls.target);
    rig.introT = 0;
  }
  if (snap) {
    // 进入 / 拖时间轴：一次到位，不做长时间横扫
    rig.anchor.copy(rigA);
    rig.yaw = yawT;
    rig.dist = clamp(R * 2.0, cfg.minD, cfg.maxD);
    if (!entering) rig.introT = 1;
  } else {
    if (rig.anchor.distanceTo(rigA) > 4000) rig.anchor.copy(rigA);
    else rig.anchor.lerp(rigA, 1 - Math.exp(-dt * cfg.anchorRate));
    rig.yaw += angDiff(rig.yaw, yawT) * (1 - Math.exp(-dt * cfg.yawRate));
  }

  // 机位相对锚点的方向：后方 az 度、上方 elev 度（不引入 roll）
  const e = cfg.elev * Math.PI / 180;
  const cy = rig.yaw + cfg.az * Math.PI / 180;
  rigMount.set(-Math.sin(cy) * Math.cos(e), Math.sin(e), -Math.cos(cy) * Math.cos(e));
  // 瞄准点先取质心，再往前（下游）带一点，给「它要去的地方」留画面
  rigAim.copy(rig.anchor)
    .addScaledVector(rigTmp.set(Math.sin(rig.yaw), 0, Math.cos(rig.yaw)), R * 0.16);

  // 闭环取景：投影包围盒 → 反解距离 → 把瞄准点收到水体中心，迭代 3 次
  let d = clamp(rig.dist, cfg.minD, cfg.maxD);
  for (let it = 0; it < 3; it++) {
    rigPos.copy(rigAim).addScaledVector(rigMount, d);
    const need = groundAt(rigPos.x, rigPos.z) + 90;
    if (rigPos.y < need) rigPos.y = need;
    camera.position.copy(rigPos);
    camera.lookAt(rigAim);
    camera.updateMatrixWorld(true);
    let x0 = 1e9, x1 = -1e9, y0 = 1e9, y1 = -1e9, n = 0;
    for (const p of pts) {
      const v = rigTmp.copy(p).project(camera);
      if (v.z > 1) continue;
      n++;
      if (v.x < x0) x0 = v.x; if (v.x > x1) x1 = v.x;
      if (v.y < y0) y0 = v.y; if (v.y > y1) y1 = v.y;
    }
    if (n < pts.length * 0.5) break;          // 一半以上不在前方，先不调
    const frac = (y1 - y0) / 2;               // 主体占画面高的比例
    const xc = (x0 + x1) / 2, yc = (y0 + y1) / 2;
    // 距离：占高比与距离近似成反比；单步限幅，防止湿点增减带来的抽动
    d = clamp(d * clamp(cfg.occ / Math.max(frac, 0.02), 0.72, 1.38), cfg.minD, cfg.maxD);
    // 瞄准点：找投影最接近包围盒中心的那个水点，向它靠
    let bw = null, bwd = 1e9;
    for (const p of pts) {
      const v = rigTmp.copy(p).project(camera);
      if (v.z > 1) continue;
      const dd = (v.x - xc) * (v.x - xc) + (v.y - yc) * (v.y - yc);
      if (dd < bwd) { bwd = dd; bw = p; }
    }
    if (bw) rigAim.lerp(bw, 0.55);
  }
  // 距离也要低通：远景/近景切换才不会有「拉风箱」的感觉
  rig.dist = lerp(rig.dist, d, 1 - Math.exp(-dt * cfg.distRate));

  rigPos.copy(rigAim).addScaledVector(rigMount, rig.dist);
  const needY = groundAt(rigPos.x, rigPos.z) + 90;
  if (rigPos.y < needY) rigPos.y = needY;

  if (rig.introT < 1) {
    // 摇臂过渡：从点下按钮那一刻的视角平滑推到跟拍机位
    rig.introT = Math.min(1, rig.introT + dt / cfg.intro);
    const k = smooth(rig.introT);
    camera.position.lerpVectors(rig.cam0, rigPos, k);
    controls.target.lerpVectors(rig.tgt0, rigAim, k);
  } else {
    camera.position.copy(rigPos);
    controls.target.copy(rigAim);
  }
}
// 播放跟拍与洪水航拍共用同一套解算，只是 rig 参数不同
function updateDisasterFollow(dt) { updateFollowRig(dt, 'follow'); }
function updateCineCamera(dt) { updateFollowRig(dt, 'cine'); }
